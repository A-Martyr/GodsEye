# --- PaddleOCR's own dependencies ------------------------------------
# _ensure_paddleocr() git-clones PaddlePaddle/PaddleOCR to make `ppocr`
# importable, but it does NOT install that repo's requirements.txt. These are
# the packages `from ppocr.postprocess import build_post_process` pulls in at
# import time, traced through the actual import chain:
#
#   ppocr.postprocess.__init__
#     -> db_postprocess     : shapely, pyclipper
#     -> ct_postprocess     : pyclipper
#     -> picodet_postprocess: scipy
#     -> pg_postprocess -> ppocr.utils.e2e_utils.pgnet_pp_utils
#                       -> extract_textpoint_slow : scikit-image
#     -> ppocr.utils.poly_nms : shapely
#
# Without them load() raises ModuleNotFoundError, swallows it, and returns
# None - which looks exactly like "the new engine silently isn't used".
shapely>=2.0 ; extra == "paddle"
pyclipper>=1.3 ; extra == "paddle"
scikit-image>=0.22 ; extra == "paddle"

# Also required by group 5, but NOT installable with pip:
#   git   - on PATH. _ensure_paddleocr() shells out to `git clone`.
